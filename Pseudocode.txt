# Sorting Algorithm (Release vs Chronological)


#The release order as well as the chronological order are not calculated. They both have their own set of values. For release order, it's the
#seperation of all the movies in their own phases (which was set by the Marvel company itself). For chronological order, I looked at all
#the lists created by others, to create my own chronological order that is accepted by most fans. The algorithm follows this predefined order and
#matches each ID to the corresponding movie in the input list (*For the shows and the X-Men movies there is no dividing into phases, so For
#release order I simply go through the actual releases)

FUNCTION sorting(list, sortmode)
    IF sortmode = "release" THEN
        RETURN ReleaseOrderSorting(list)
    ELSE
        RETURN ChronologicalOrderSorting(list)
END FUNCTION

#Percentage Algorithms

FUNCTION calculateMoviesPercentage(watchedMovies, totalMovies)
    percentage = (watchedMovies/totalMovies) * 100
    RETURN ROUND(percentage)

END FUNCTION

#LocalStorage Algorithms

FUNCTION saveProgress(watchedMovies, watchedEpisodes, rewatches)
    data = {
        movies: watchedMovies,
        episodes: watchedEpisodes,
        rewatches: rewatches
    }

    jsonString = CONVERT data to jsonString

    STORE jsonString IN LocalStorage UNDER "progress"
END FUNCTION

FUNCTION loadProgress()
    saved = READ "progress" FROM LocalStorage

    IF saved IS NULL THEN
        RETURN {
            movies: EMPTY LIST,
            episodes: EMPTY OBJECT
            rewatches: EMPTY OBJECT
        }
    TRY
        data = PARSE saved
        RETURN data
    CATCH error
        RETURN {
            movies: EMPTY LIST,
            episodes: EMPTY OBJECT
            rewatches: EMPTY OBJECT    
        }
END FUNCTION
